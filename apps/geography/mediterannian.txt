    <div class="game-container">
        <div class="countries-container" id="draggable-container">
            <div class="country-label" draggable="true" data-id="israel">ישראל</div>

            <div class="drop-zone" id="drop-italy" data-target="italy"></div>

        <div class="success-message" id="success-msg">כל הכבוד! השלמת את המפה בהצלחה! 🎉</div>
    </div>

    <script>
        const draggables = document.querySelectorAll('.country-label');
        const dropZones = document.querySelectorAll('.drop-zone');
        let correctCount = 0;
        const totalCountries = draggables.length;

        // אירועי גרירה עבור התוויות
        draggables.forEach(draggable => {
            draggable.addEventListener('dragstart', (e) => {
                e.dataTransfer.setData('text/plain', draggable.dataset.id);
                draggable.style.opacity = '0.5';
            });

            draggable.addEventListener('dragend', () => {
                draggable.style.opacity = '1';
            });
        });

        // אירועי שחרור עבור נקודות על המפה
        dropZones.forEach(zone => {
            zone.addEventListener('dragover', (e) => {
                e.preventDefault(); // מאפשר את השחרור
            });

            zone.addEventListener('drop', (e) => {
                e.preventDefault();
                const draggedId = e.dataTransfer.getData('text/plain');
                const targetId = zone.dataset.target;

                // בדיקה האם המדינה נגררה למקום הנכון
                if (draggedId === targetId && !zone.classList.contains('correct')) {
                    const draggedElement = document.querySelector(`.country-label[data-id="${draggedId}"]`);
                    
                    // שינוי עיצוב הנקודה לשם המדינה
                    zone.classList.add('correct');
                    zone.textContent = draggedElement.textContent;
                    
                    // העלמת התווית המקורית
                    draggedElement.style.display = 'none';
                    correctCount++;

                    // בדיקת ניצחון
                    if (correctCount === totalCountries) {
                        document.getElementById('success-msg').style.display = 'block';
                    }
                }
            });
        });
    </script>
